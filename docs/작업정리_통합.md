### 6-2. 위젯 수정 패턴 (편집기 탭에서 실행)
```js
// A. 컨테이너가 잡힐 때
var c=elementor.getContainer('위젯ID');
$e.run('document/elements/settings',{container:c,settings:{html:t /*text-editor는 editor:t*/},options:{external:true}});
$e.internal('document/save/set-is-modified',{status:true});
$e.run('document/save/default').then(function(){window.__sv='saved';});
// B. getContainer가 throw / 위젯에 id가 없을 때 — Backbone 모델 직접
var target=null;(function walk(coll){coll.each(function(m){var s=m.get('settings'),v=s&&s.get?s.get('editor'):null;
 if(typeof v==='string'){if(v.indexOf('고유문자열')>-1)target=m;} var ch=m.get('elements'); if(ch){if(ch.length)walk(ch);}});})(elementor.elements);
```
